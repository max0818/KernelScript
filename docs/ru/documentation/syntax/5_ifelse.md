## Условные операторы

## if-elif-else

## Синтаксис
```ks
if (значение1) {
	код1
} elif (значение2) {
	код2
} elif (значение3) {
	код3
} else {
	код4
}
```

## match-case

## Синтаксис
```ks
match (значение) {
	case (значение1) {
		код1
	}

	case (значение2) {
		код2
	}

	case (значение3) {
		код3
	}

	case _ {
		код4
	}
}
```

## Тернарный оператор

## Синтаксис
```ks
значение ? значение1 : значение2
```

## Примеры

### Проверка числа
```ks
var x: int = 5

if (x < 5) {
	x++;
} elif (x > 5) {
	x--;
} else {
	x = 0
}
```

### Проверка строки
```ks
var y: string = 'text'

match (y) {
	case ('name') {
		y += 1
	}

	case ('text') {
		y = 'name'
	}

	case _ {
		y = null
	}
}
```

### Маленькая проверка
```ks
const x = 5

var text: string = x < 5 ? "name" : "text" + x
```
